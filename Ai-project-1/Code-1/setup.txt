def neuron(x, w, b):
    return 1 if (x * w + b) > 0 else 0

data = [
    (1, 0),
    (2, 0),
    (5, 1),
    (8, 1),
]

w = 0.1
b = 0

for epoch in range(20):
    for x, target in data:
        y = neuron(x, w, b)
        error = target - y
        w += error * 0.1
        b += error * 0.1

print("Number prediction results 7:", neuron(7, w, b))

import math
import random

def sigmoid(x):
    return 1 / (1 + math.exp(-x))

def neuron(x1, x2, w1, w2, b):

    z = (
        x1 * w1 +
        x2 * w2 +
        b
    )

    return sigmoid(z)

data = [
    ((1, 1), 0),
    ((2, 1), 0),
    ((3, 2), 0),
    ((4, 2), 0),

    ((5, 5), 1),
    ((6, 5), 1),
    ((7, 6), 1),
    ((8, 7), 1),
]

learning_rate = 0.1

w1 = random.uniform(-1, 1)
w2 = random.uniform(-1, 1)

b = random.uniform(-1, 1)

best_loss = float("inf")

for epoch in range(1000):

    total_loss = 0

    for (x1, x2), target in data:

        y = neuron(
            x1,
            x2,
            w1,
            w2,
            b
        )

        error = target - y

        loss = error ** 2

        total_loss += loss

        gradient = error * y * (1 - y)

        w1 += (
            learning_rate *
            gradient *
            x1
        )

        w2 += (
            learning_rate *
            gradient *
            x2
        )

        b += (
            learning_rate *
            gradient
        )

    if total_loss < best_loss:

        best_loss = total_loss

        best_w1 = w1
        best_w2 = w2
        best_b = b

    if epoch % 100 == 0:

        print(
            f"epoch={epoch} "
            f"loss={total_loss:.6f}"
        )

w1 = best_w1
w2 = best_w2
b = best_b

print("\nTraining completed")

print("w1 =", w1)
print("w2 =", w2)
print("b  =", b)

correct = 0

for (x1, x2), target in data:

    y = neuron(
        x1,
        x2,
        w1,
        w2,
        b
    )

    prediction = (
        1
        if y >= 0.5
        else 0
    )

    if prediction == target:
        correct += 1

accuracy = correct / len(data)

print("\nAccuracy =", accuracy)

print("\nPredictions")

for x1, x2 in [

    (2, 1),
    (3, 2),
    (5, 5),
    (7, 6),
    (10, 10),

]:

    y = neuron(
        x1,
        x2,
        w1,
        w2,
        b
    )

    prediction = (
        1
        if y >= 0.5
        else 0
    )

    confidence = y * 100

    print(
        f"x1={x1} "
        f"x2={x2} "
        f"prob={y:.4f} "
        f"class={prediction} "
        f"confidence={confidence:.1f}%"
    )
